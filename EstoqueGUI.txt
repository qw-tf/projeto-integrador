import javax.swing.JOptionPane;

public class EstoqueGUI {

    private Estoque estoque = new Estoque();

    public void exibirMenu() {
        String[] opcoes = {"Adicionar Produto", "Excluir Produto", "Listar Produtos", "Sair"};
        int opcao;

        do {
            opcao = JOptionPane.showOptionDialog(
                null,
                "Escolha uma opção:",
                "Sistema de Estoque",
                JOptionPane.DEFAULT_OPTION,
                JOptionPane.PLAIN_MESSAGE,
                null,
                opcoes,
                opcoes[0]
            );

            switch (opcao) {
                case 0 -> adicionarProdutoGUI();
                case 1 -> excluirProdutoGUI();
                case 2 -> listarProdutosGUI();
                default -> {} // Nenhuma ação para "Sair"
            }
        } while (opcao != 3); // enquanto não clicar em "Sair"
    }

    private void adicionarProdutoGUI() {
        try {
            String nome = JOptionPane.showInputDialog("Nome do produto:");
            if (nome == null) return;

            int quantidade = Integer.parseInt(JOptionPane.showInputDialog("Quantidade:"));
            double preco = Double.parseDouble(JOptionPane.showInputDialog("Preço:"));

            int opcaoPerecivel = JOptionPane.showConfirmDialog(null, "O produto é perecível?", "Tipo", JOptionPane.YES_NO_OPTION);
            boolean perecivel = (opcaoPerecivel == JOptionPane.YES_OPTION);

            String dataValidade = "";
            if (perecivel) {
                dataValidade = JOptionPane.showInputDialog("Data de validade (dd/mm/yyyy):");
            }

            estoque.adicionarProduto(nome, quantidade, preco, perecivel, dataValidade);

            JOptionPane.showMessageDialog(null, "Produto adicionado com sucesso!");

        } catch (NumberFormatException e) {
            JOptionPane.showMessageDialog(null, "Erro de formato numérico. Operação cancelada.");
        } catch (ValidacaoException e) {
            JOptionPane.showMessageDialog(null, e.getMessage());
        }
    }

    private void excluirProdutoGUI() {
        try {
            int codigo = Integer.parseInt(JOptionPane.showInputDialog("Digite o código do produto a excluir:"));

            Produto produto = estoque.buscarProduto(codigo);
            if (produto == null) {
                JOptionPane.showMessageDialog(null, "Produto não encontrado.");
                return;
            }

            int confirm = JOptionPane.showConfirmDialog(
                null,
                "Deseja remover o produto: " + produto.getNome() + "?",
                "Confirmar Exclusão",
                JOptionPane.YES_NO_OPTION
            );

            if (confirm == JOptionPane.YES_OPTION) {
                estoque.excluirProduto(codigo);
                JOptionPane.showMessageDialog(null, "Produto removido com sucesso.");
            }

        } catch (NumberFormatException e) {
            JOptionPane.showMessageDialog(null, "Código inválido.");
        } catch (ValidacaoException e) {
            JOptionPane.showMessageDialog(null, e.getMessage());
        }
    }

    private void listarProdutosGUI() {
        StringBuilder lista = new StringBuilder();

        if (estoque.getProdutos().isEmpty()) {
            lista.append("Nenhum produto cadastrado.");
        } else {
            for (Produto p : estoque.getProdutos()) {
                lista.append("Código: ").append(p.getCodigo())
                     .append(", Nome: ").append(p.getNome())
                     .append(", Quantidade: ").append(p.getQuantidade())
                     .append(", Preço: ").append(p.getPreco());

                if (p instanceof ProdutoPerecivel perecivel) {
                    lista.append(", Validade: ").append(perecivel.getDataDeValidade());
                }

                lista.append("\n");
            }
        }

        JOptionPane.showMessageDialog(null, lista.toString(), "Lista de Produtos", JOptionPane.INFORMATION_MESSAGE);
    }

        public static void main(String[] args) {
            EstoqueGUI gui = new EstoqueGUI();
            gui.exibirMenu();
        }

    }
